=== LH_Charge : point de terminaison SQL ===

SELECT station, sessions, failed, failure_rate, energy_kwh, revenue_eur, avg_minutes, alert FROM station_kpi ORDER BY failure_rate DESC;

SELECT 'sessions_raw' AS table_name, COUNT(*) AS lignes FROM sessions_raw UNION ALL SELECT 'sessions', COUNT(*) FROM sessions UNION ALL SELECT 'station_kpi', COUNT(*) FROM station_kpi;
